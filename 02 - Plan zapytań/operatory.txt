	soh.SalesOrderID, soh.OrderDate, soh.CustomerID,
	sod.ProductID, sod.OrderQty, sod.UnitPrice

/*  Merge Join  */

SELECT 	c.*
FROM Sales.Customer AS c
JOIN Sales.SalesOrderHeader AS soh ON soh.CustomerID = c.CustomerID

/*  Hash Match (Join)  */
-- Hash Match (Inner Join)
-- tworzona jest tabela haszująca dla mniejszej tabeli (w bazie tempdb)
-- potrafi być bardzo wydajny w przypadku dwóch bardzo dużych tabel

SELECT pp.ProductID, pp.Name, ps.Name
FROM Production.Product pp
JOIN Production.ProductSubcategory ps
ON ps.ProductSubcategoryID = pp.ProductSubcategoryID

-- pomijanie złączenia
SELECT pp.*
FROM Production.Product pp LEFT JOIN Production.ProductSubcategory ps
		ON ps.ProductSubcategoryID = pp.ProductSubcategoryID
