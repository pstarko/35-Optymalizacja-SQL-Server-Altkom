CREATE OR ALTER FUNCTION dbo.GetLangName (@langid int)
RETURNS nvarchar(100)
AS 
BEGIN
	RETURN (SELECT TOP 1 alias FROM sys.syslanguages WHERE msglangid = @langid)
END
GO

FROM msgs

FROM msgs
