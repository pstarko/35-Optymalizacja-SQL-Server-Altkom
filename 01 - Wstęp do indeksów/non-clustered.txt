/*
	NONCLUSTERED INDEX
	INCLUDE
	
	Jeśli kolumna nie znajduje się w WHERE/JOIN/GROUP BY/ORDER BY, ale tylko na liście kolumn w klauzuli SELECT, 
	używasz INCLUDE.

	Klauzula INCLUDE dodaje dane na najniższym poziomie/liściu, 
	a nie w drzewie indeksu. Powoduje to, że indeks jest mniejszy, ponieważ nie jest częścią drzewa

	Kolumny INCLUDE nie są kluczowymi kolumnami w indeksie, więc nie są uporządkowane. 
	Oznacza to, że tak naprawdę nie jest przydatny do predykatów, sortowania itp., 
	Jak wspomniałem powyżej. Jednak może to być przydatne, jeśli masz wyszukiwanie resztkowe w kilku wierszach 
	z kluczowych kolumn
*/
CREATE NONCLUSTERED INDEX IDX_NCL_AllData_City ON AllData (City)
INCLUDE (Firstname, Lastname)
WITH DROP_EXISTING
